# loop10 -- counted loop, 10 iterations.
# Self-checking: sums 1..10 and requires 55. Leaves x1 = 1 on pass.
# The ecall must be the last executed instruction: anything after it retires
# during the testbench's drain window, so the trailing instructions are nops.
        addi x5, x0, 10
        addi x8, x0, 0
loop:   add  x8, x8, x5
        addi x5, x5, -1
        bne  x5, x0, loop
        addi x1, x0, 0           # default: fail
        addi x7, x0, 55
        bne  x8, x7, done        # wrong total -> leave x1 = 0
        addi x1, x0, 1           # pass
done:   ecall
        nop
        nop
        nop
        nop
        nop
        nop
