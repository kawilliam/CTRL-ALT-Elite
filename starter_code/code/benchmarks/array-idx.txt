# array-idx -- sum a[i] using a scaled index, the shape a compiler emits for
# `for (i=0;i<n;i++) s += a[i];`
#
# Loop body, 6 instructions with one adjacent add -> lw pair:
#   slli x10, x9, 2      byte offset = i*4
#   add  x11, x2, x10    address   = base + offset   <-- add
#   lw   x12, 0(x11)     value                        <-- lw   (fusable pair)
#   add  x8, x8, x12     sum += value
#   addi x9, x9, 1       i++
#   blt  x9, x6, loop
        lui  x2, 1               # x2 = 0x1000, array base
        addi x5, x0, 1
        sw   x5, 0(x2)
        addi x5, x0, 2
        sw   x5, 4(x2)
        addi x5, x0, 3
        sw   x5, 8(x2)
        addi x5, x0, 4
        sw   x5, 12(x2)
        addi x5, x0, 5
        sw   x5, 16(x2)
        addi x5, x0, 6
        sw   x5, 20(x2)
        addi x5, x0, 7
        sw   x5, 24(x2)
        addi x5, x0, 8
        sw   x5, 28(x2)
        addi x8, x0, 0           # sum
        addi x9, x0, 0           # i
        addi x6, x0, 8           # n
loop:   slli x10, x9, 2          # byte offset
        add  x11, x2, x10        # address  (add)
        lw   x12, 0(x11)         # value    (lw)  <- fusable pair
        add  x8, x8, x12         # sum += a[i]
        addi x9, x9, 1           # i++
        blt  x9, x6, loop
        addi x1, x0, 0           # default: fail
        addi x7, x0, 36          # 1+2+...+8
        bne  x8, x7, done
        addi x1, x0, 1           # pass
done:   ecall
        nop
        nop
        nop
        nop
        nop
        nop
